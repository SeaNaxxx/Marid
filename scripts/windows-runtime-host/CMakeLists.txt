cmake_minimum_required(VERSION 3.20)
project(qodex_runtime_host LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(qodex-runtime-host main.cpp)
add_executable(qodex-runtime-host-acceptance EXCLUDE_FROM_ALL acceptance.cpp)
add_executable(qodex-installer-process-helper installer-process-helper.cpp)
add_executable(qodex-installer-process-fixture EXCLUDE_FROM_ALL installer-process-fixture.cpp)
target_link_libraries(qodex-installer-process-fixture PRIVATE advapi32)
foreach(target qodex-runtime-host qodex-runtime-host-acceptance qodex-installer-process-helper qodex-installer-process-fixture)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /utf-8)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra)
    target_link_options(${target} PRIVATE -municode -static)
  endif()
endforeach()
